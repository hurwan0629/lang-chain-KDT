# 버블 정렬
버블 정렬은 일반적으로 실무에서 잘 쓰이지 않는 정렬 방식으로 교육을 위해 많이 쓰이는 방식입니다.

정렬의 이름에 존재하는 **버블**이라는 말과 같이 리스트의 앞부분부터 2개씩 순서대로 `[0, 1] [1, 2] [2, 3], ... [n-2, n-1]`을 정렬하는 것을 반복하여 최종적으로 정렬 상태를 만들게 됩니다.

최초의 실행에는 확실하게 `n-1`번째 인덱스가 제자리를 찾고, 그다음 실행에는 `n-2`가 확정되어 최종적으로 `n*(n-1) / 2`의 시간 복잡도, 즉 `O(n^2)`의 시간 복잡도를 가지며 공간 복잡도의 경우에는 그나마 교환을 위한 임시 저장 공간만을 요구하기 때문에 `O(1)`의 공간 복잡도를 소모합니다.

최선의 경우에는 `O(n)`으로 그냥 한 번 순회했을 때 정렬되어 있으면 그냥 패스하기도 합니다.

> 정렬에서 `O(n^2)`라는 시간 복잡도는 웬만하면 잘 나타나지 않는 최악에 가까운 시간 복잡도입니다.

# 버블 정렬의 구현

import random

# 20 ~ 30개 리스트 (중복이 많으면 더 줄 수 있음)
# 1 ~ 2000 중 랜덤 고르기
rand_list = random.sample([a for a in range(1, 2001)], k=random.randint(20, 30))
print(rand_list)


def bubble_sort(rand_list:list, order_type: str):
  # 횟수 세기용 count
  count = 0

  # 버블 정렬
  # 마지막 한 번은 안 해도 문제없음
  for i in range(len(rand_list)-1):
    # 인덱스를 0 ~ (n-1)-i 까지 순회하며 정렬
    # 추가로 n-2, n-1 인덱스 교환만 하면 되기 때문에 n-2까지만 가면 됨
    
    # 중간에 정렬되었을 경우를 참고해서 한 번 검사해 주기
    swap = 0
    # print(f"{i+1}/{len(rand_list)-1}번째 순회")
    for j in range(len(rand_list) - i - 1):
      count += 1
      if rand_list[j] > rand_list[j+1] if order_type == "asc" else rand_list[j] < rand_list[j+1]:
          swap += 1
          # ▼ ▼ ▼ ▼ ▼ ▼ 파이썬의 좋은 점 ▼ ▼ ▼ ▼ ▼ ▼ 
          # 함수 하나를 안 만들어도 됨...
          rand_list[j], rand_list[j+1] = rand_list[j+1], rand_list[j]
    
    if swap == 0:
      print("정렬 완료")
      break

  print(rand_list)
  print(f"rand_list 길이: {len(rand_list)}")
  print(f"버블 횟수: {count}")
  print(f"N^2 = {len(rand_list)**2}")

print("\n --- asc --- \n")
bubble_sort(rand_list, "asc")
print("\n --- desc --- \n")
bubble_sort(rand_list, "desc")

# 선택 정렬과 삽입 정렬
선택 정렬과 삽입 정렬은 메커니즘이 닮아 있습니다.

선택 정렬은 뒤의 값들에서 최솟값을 하나씩 앞으로 보내는 방식이며 삽입 정렬은 앞에서부터 하나씩 올바른 위치에 놓는 방식입니다. 

예를 들어

`d a b e c`가 있으면 

선택 정렬의 경우에는
```
[a] [d b e c]
[a b] [d e c]
[a b c] [d e]
[a b c d] [e]
[a b c d e]
```
방식이고 삽입 정렬은
```
[d] [a b e c]
[a d] [b e c]
[a b d] [e c]
[a b d e] [c]
[a b c d e]
```
입니다.

결국 모두 랜덤값 기준 `O(n^2)`의 시간 복잡도를 가지지만 비운의 교육용 3대 정렬 알고리즘인 `버블`, `삽입`, `선택` 중에서 `삽입 정렬`은 이미 정렬되어 있는 경우 한 번의 순회만으로 추가 값을 넣을 수 있다는 점에서 실무에서 종종 쓰이기도 합니다.

> 역시나 잘 쓰이지 않는다는 점은 존재하지만 선택정렬 또한 **교환 횟수가 적다**라는 이점이 존재하여 교환 비용이 비싼 경우에는 유리합니다.

# 삽입 정렬 구현
import random

# 20~30개 리스트 
# 1 ~ 2000 중 랜덤 고르기
rand_list = random.sample([a for a in range(1, 2001)], k=random.randint(20, 30))
print(rand_list)

order_type = "asc"
# order_type = "desc" # (desc가 아니어도 asc만 아니면 됨)

# 2번 만들기 귀찮아서.... 하핫
def insertion_sort(rand_list:list, order_type: str) -> list:
  # 횟수 세기용 count
  count = 0
  # 처음부터 하나씩 골라서 앞에 보내는 것이기 때문에 len(rand_list)-1까지 반복.
  # 하지만 0의 경우에는 이미 정렬되었다 가정. 따라서 1부터 시작
  for i in range(1, len(rand_list)):
    # print(f"{i}/{len(rand_list)-1}번째 순회")
    # print(rand_list)
    
    # 다음 값 뽑기
    pick = rand_list[i]
    j = i - 1
    # 기존 앞부분에서 알맞은 곳 탐색
    # 맨 뒤에서부터 자기의 알맞은 위치 찾아가기
    while j >= 0 and (rand_list[j] > pick if order_type == "asc" else rand_list[j] < pick):
      count+=1
      j -= 1
    
    # j+1 위치에 값을 넣으면 됨 (rand_list[j] < pick 인 상황이기 때문)
    # 나머지는 다 오른쪽으로 밀어버리기
    rand_list[j+2:i+1], rand_list[j+1] = rand_list[j+1:i], pick

  print(rand_list)
  print(f"rand_list 길이: {len(rand_list)}")
  print(f"탐색 횟수: {count}")
  print(f"N^2 = {len(rand_list)**2}")
  return rand_list

print("\n --- asc --- \n")
insertion_sort(rand_list, "asc")
print("\n --- desc --- \n")
insertion_sort(rand_list, "desc")

# 선택 정렬 구현
import random

# 20 ~ 30개 리스트 (중복이 많으면 더 줄 수 있음)
# 1 ~ 2000 중 랜덤 고르기
rand_list = random.sample([a for a in range(1, 2001)], k=random.randint(20, 30))
print(rand_list)

def selection_sort(rand_list:list, order_type: str) -> list:
  # 횟수 세기용 count
  count = 0

  # 무조건 순회 n*(n+1)/2
  for i in range(len(rand_list)):
    # print(f"{i+1}/{len(rand_list)-1}번째 순회")
    # print(rand_list)
    
    if order_type == "asc":
      # i번째 인덱스 값 구하기
      # 방법은 i부터 최솟값 탐색
      min = rand_list[i]
      min_index = i
      for j in range(i, len(rand_list)):
        count+=1
        if min > rand_list[j]:
          min = rand_list[j]
          min_index = j
      
      rand_list[i], rand_list[min_index] = min, rand_list[i]
    else:
      # i번째 인덱스 값 구하기
      # 방법은 i부터 최댓값 탐색
      max = rand_list[i]
      max_index = i
      for j in range(i, len(rand_list)):
        count+=1
        if max < rand_list[j]:
          max = rand_list[j]
          max_index = j
      
      rand_list[i], rand_list[max_index] = max, rand_list[i]

  print(rand_list)
  print(f"rand_list 길이: {len(rand_list)}")
  print(f"탐색 횟수: {count}")
  print(f"N^2 = {len(rand_list)**2}")
  return rand_list

print("\n --- asc --- \n")
selection_sort(rand_list, "asc")
print("\n --- desc --- \n")
selection_sort(rand_list, "desc")
