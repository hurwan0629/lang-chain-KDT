class ExampleBox():

  def __init__(self, prev, next, data):
    self.prev = prev
    self.curr = self
    self.next = next
    self.data = data

  def append(self, data):
    # 내가 맨 뒤면
    if self.next is None:
      self.next = ExampleBox(self, None, data)
    # 앞에 칸이 있으면 연결해주기
    else:
      # 기존 앞칸과 새 칸 연결해주기. (이때 내 prev는 아직 기존 앞칸과 연결되어있음)
      self.next.prev = ExampleBox(self, self.next, data)
      # 내 앞칸을 기존 앞칸의 next와 연결해주기
      self.next = self.next.prev

    return self.next

  def prepend(self, data):
    # 만약에 해당 박스가 맨 앞의 박스라면
    if self.prev is None:
      self.prev = ExampleBox(None, self, data)
    # 앞에 칸이 있으면 연결해주기
    else:
      # 기존 앞칸과 새 칸 연결해주기. (이때 내 prev는 아직 기존 앞칸과 연결되어있음)
      self.prev.next = ExampleBox(self.prev, self, data)
      # 내 앞칸을 기존 앞칸의 next와 연결해주기
      self.prev = self.prev.next

    return self.prev

  def __next__(self):
    if self.next is not None:
      return self.next
    raise StopIteration

직접 박스 연결하기

ExampleBox

다음으로 앞 칸과 뒤 칸을 기억하는 ExampleBox를 만들었습니다. 각 박스는 prev, next, data를 가지고 있고, append()와 prepend()로 중간에 새 박스를 연결합니다.

ExampleBox 코드 보기 
class ExampleBox():

  def __init__(self, prev, next, data):
    self.prev = prev
    self.curr = self
    self.next = next
    self.data = data

  def append(self, data):
    # 내가 맨 뒤면
    if self.next is None:
      self.next = ExampleBox(self, None, data)
    # 앞에 칸이 있으면 연결해주기
    else:
      # 기존 앞칸과 새 칸 연결해주기. (이때 내 prev는 아직 기존 앞칸과 연결되어있음)
      self.next.prev = ExampleBox(self, self.next, data)
      # 내 앞칸을 기존 앞칸의 next와 연결해주기
      self.next = self.next.prev

    return self.next

  def prepend(self, data):
    # 만약에 해당 박스가 맨 앞의 박스라면
    if self.prev is None:
      self.prev = ExampleBox(None, self, data)
    # 앞에 칸이 있으면 연결해주기
    else:
      # 기존 앞칸과 새 칸 연결해주기. (이때 내 prev는 아직 기존 앞칸과 연결되어있음)
      self.prev.next = ExampleBox(self.prev, self, data)
      # 내 앞칸을 기존 앞칸의 next와 연결해주기
      self.prev = self.prev.next

    return self.prev

  def __next__(self):
    if self.next is not None:
      return self.next
    raise StopIteration

iter와 yield

BoxSpace를 반복 가능하게 만들기

BoxSpace에서는 시작점을 저장하고, __iter__() 안에서 현재 박스의 data를 yield한 뒤 다음 박스로 이동하게 만들었습니다.

BoxSpace 코드 보기 
class BoxSpace():

  def __init__(self, data):
    self.starting_point = ExampleBox(None, None, data)

  def __iter__(self):
    curr = self.starting_point

    while curr is not None:
      yield curr.data
      curr = curr.next

정리

확인한 것 내용
set 순서를 기준으로 쓰는 자료형은 아니지만 for문으로 순회할 수 있습니다.
ExampleBox prev와 next를 저장해서 앞뒤 박스를 연결하는 구조를 만들었습니다.
__iter__() 객체가 iter()에 들어갔을 때 어떤 방식으로 값을 하나씩 돌려줄지 정했습니다.
yield 리스트를 한 번에 만들지 않고, 반복 과정에서 값을 하나씩 넘겨주도록 사용했습니다.
str 문자열도 반복 가능한 객체라서 문자 단위로 꺼낼 수 있습니다.
이번 실습은 for문이 단순히 리스트만 도는 문법이 아니라, 반복 가능한 객체와 약속된 메서드를 이용해 값을 하나씩 꺼내는 구조라는 점을 확인하는 과정이었습니다.
